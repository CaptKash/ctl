   102	
   103	export default function CarDetailScreen() {
   104	  const { id } = useLocalSearchParams<{ id: string }>();
   105	  const carId = parseInt(id ?? "0");
   106	  const insets = useSafeAreaInsets();
   107	  const C = useColors();
   108	  const { i18n } = useTranslation();
   109	  const { unit } = useDistanceUnit();
   110	  const qc = useQueryClient();
   111	  const topPad = Platform.OS === "web" ? 67 : insets.top;
   112	
   113	  const [confirmModal, setConfirmModal] = useState<{
   114	    visible: boolean;
   115	    title: string;
   116	    message: string;
   117	    onConfirm: () => void;
   118	  }>({ visible: false, title: "", message: "", onConfirm: () => {} });
   119	
   120	  const [detailEvent, setDetailEvent] = useState<EventItem | null>(null);
   121	  const [photoViewer, setPhotoViewer] = useState<string | null>(null);
   122	  const [downloadingPhoto, setDownloadingPhoto] = useState(false);
   123	  const handleDownloadPhoto = (url: string) => {
   124	    if (!url || downloadingPhoto) return;
   125	    setPhotoViewer(null);
   126	    setTimeout(() => doShare(url), 600);
   127	  };
   128	
   129	  const doShare = async (url: string) => {
   130	    setDownloadingPhoto(true);
   131	    try {
   132	      const filename = `ctl_photo_${Date.now()}.jpg`;
   133	      const fileUri = `${FileSystem.cacheDirectory}${filename}`;
   134	
   135	      if (url.startsWith("data:")) {
   136	        const base64 = url.split(",")[1];
   137	        await FileSystem.writeAsStringAsync(fileUri, base64, {
   138	          encoding: "base64",
   139	        });
   140	      } else {
   141	        await FileSystem.downloadAsync(url, fileUri);
   142	      }
   143	
   144	      const available = await Sharing.isAvailableAsync();
   145	      if (!available) throw new Error("Sharing not available on this device");
   146	
   147	      await Sharing.shareAsync(fileUri, {
   148	        mimeType: "image/jpeg",
   149	        UTI: "public.jpeg",
   150	        dialogTitle: "Save Photo",
   151	      });
   152	    } catch (e: unknown) {
   153	      const msg = e instanceof Error ? e.message : String(e);
   154	      Alert.alert("Save failed", msg);
   155	    } finally {
   156	      setDownloadingPhoto(false);
   157	    }
   158	  };
   159	
   160	  const carQuery = useQuery<Car>({
   161	    queryKey: ["car", carId],
   162	    queryFn: () => apiGet<Car>(`/cars/${carId}`),
   163	  });
   164	
   165	  const faultsQuery = useQuery<FaultRecord[]>({
   166	    queryKey: ["malfunctions", carId],
   167	    queryFn: () => apiGet<FaultRecord[]>(`/cars/${carId}/malfunctions`),
   168	  });
   169	
   170	  const maintenanceQuery = useQuery<MaintenanceRecord[]>({
   171	    queryKey: ["maintenance", carId],
   172	    queryFn: () => apiGet<MaintenanceRecord[]>(`/cars/${carId}/maintenance`),
   173	  });
   174	
   175	  const inspectionsQuery = useQuery<InspectionRecord[]>({
   176	    queryKey: ["inspections", carId],
   177	    queryFn: () => apiGet<InspectionRecord[]>(`/cars/${carId}/inspections`),
   178	  });
   179	
   180	  const events: EventItem[] = React.useMemo(() => {
   181	    const list: EventItem[] = [];
   182	    (faultsQuery.data ?? [])
   183	      .filter((r) => !r.completed)
   184	      .forEach((r) => list.push({ kind: "fault", data: r }));
   185	    (maintenanceQuery.data ?? []).forEach((r) => list.push({ kind: "maintenance", data: r }));
   186	    (inspectionsQuery.data ?? []).forEach((r) => list.push({ kind: "inspection", data: r }));
   187	    return list.sort((a, b) => b.data.date.localeCompare(a.data.date));
   188	  }, [faultsQuery.data, maintenanceQuery.data, inspectionsQuery.data]);
   189	
   190	  const isLoading = faultsQuery.isLoading || maintenanceQuery.isLoading || inspectionsQuery.isLoading;
   191

   193	    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);

   195	      visible: true,
   196	      title: "Delete Record",
   197	      message: "This record will be permanently deleted.",
   198	      onConfirm: async () => {
   199	        setConfirmModal((s) => ({ ...s, visible: false }));
   200	        await apiDelete(path);
   201	        qc.invalidateQueries({ queryKey: [queryKey, carId] });
   202	      },
   203	    });
   204	  };
   205	
   206	  const renderEvent = (ev: EventItem) => {
   207	    const meta = EVENT_META[ev.kind];
   208	    const key = `${ev.kind}-${ev.data.id}`;
   209	
   210	    if (ev.kind === "fault") {
   211	      const r = ev.data;
   212	      return (
   213	        <View key={key} style={[styles.card, { backgroundColor: C.card, borderColor: meta.color }]}>
   214	          <View style={styles.cardHeader}>
   215	            <View style={[styles.badge, { backgroundColor: meta.bg }]}>
   216	              <Feather name={meta.icon} size={11} color={meta.color} />
   217	              <Text style={[styles.badgeText, { color: meta.color }]}>{meta.label}</Text>
   218	            </View>
   219	            <Text style={[styles.dateText, { color: C.textTertiary }]}>{formatDate(r.date)}</Text>
   220	          </View>
   221	          <Text style={[styles.cardTitle, { color: C.text }]} numberOfLines={3}>{r.description}</Text>
   222	          {!!r.warningMessage && (
   223	            <Text style={[styles.warningMsgText, { color: C.textSecondary }]} numberOfLines={2}>
   224	              <Text style={styles.warningMsgLabel}>Dashboard message: </Text>
   225	              {r.warningMessage}
   226	            </Text>
   227	          )}
   228	          {!!r.dashboardMessage && (() => {
   229	            const ids = r.dashboardMessage.split(",").map((s) => s.trim()).filter(Boolean);
   230	            const lights = DASHBOARD_LIGHTS.filter((l) => ids.includes(l.id));
   231	            if (!lights.length) return null;
   232	            return (
   233	              <View style={styles.iconsRow}>
   234	                {lights.map((l) => (
   235	                  <DashboardIcon key={l.id} id={l.id} color={l.warningColor} size={20} />
   236	                ))}
   237	              </View>
   238	            );
   239	          })()}
   240	          <View style={styles.cardFooter}>
   241	            <View style={{ flex: 1 }} />
   242	            <View style={styles.cardActions}>
   243	              <Pressable
   244	                onPress={() => router.push({
   245	                  pathname: "/maintenance/form",
   246	                  params: { carId: String(carId), faultDescription: r.description, faultId: String(r.id) },
   247	                } as any)}
   248	                style={[styles.actionBtn, { backgroundColor: meta.bg, borderColor: meta.color }]}
   249	              >
   250	                <Feather name="tool" size={14} color={meta.color} />
   251	              </Pressable>
   252	              <Pressable
   253	                onPress={() => confirmDelete(`/cars/${carId}/malfunctions/${r.id}`, "malfunctions")}
   254	                style={[styles.actionBtn, { backgroundColor: meta.bg, borderColor: meta.color }]}
   255	              >
   256	                <Feather name="trash-2" size={14} color={meta.color} />
   257	              </Pressable>
   258	            </View>
   259	          </View>
   260	        </View>
   261	      );
   262	    }
   263	
   264	    if (ev.kind === "maintenance") {
   265	      const r = ev.data;
   266	      return (
   267	        <Pressable
   268	          key={key}
   269	          onPress={() => { Haptics.selectionAsync(); setDetailEvent(ev); }}
   270	          style={({ pressed }) => [styles.card, { backgroundColor: C.card, borderColor: meta.color, opacity: pressed ? 0.9 : 1 }]}
   271	        >
   272	          <View style={styles.cardHeader}>
   273	            <View style={[styles.badge, { backgroundColor: meta.bg }]}>
   274	              <Feather name={meta.icon} size={11} color={meta.color} />
   275	              <Text style={[styles.badgeText, { color: meta.color }]}>{r.type || meta.label}</Text>
   276	            </View>
   277	            <Text style={[styles.dateText, { color: C.textTertiary }]}>{formatDate(r.date)}</Text>
   278	          </View>
   279	          <Text style={[styles.cardTitle, { color: C.text }]} numberOfLines={3}>{r.description}</Text>
   280	          {(r.shop || r.cost != null) && (
   281	            <View style={styles.cardMeta}>
   282	              {r.shop && (
   283	                <>
   284	                  <Feather name="map-pin" size={12} color={C.textTertiary} />
   285	                  <Text style={[styles.metaText, { color: C.textSecondary }]}>{r.shop}</Text>
   286	                </>
   287	              )}
   288	              {r.cost != null && (
   289	                <>
   290	                  {r.shop && <View style={styles.metaDot} />}
   291	                  <Text style={[styles.metaText, { color: C.textSecondary }]}>${r.cost.toFixed(2)}</Text>
   292	                </>
   293	              )}
   294	            </View>
   295	          )}
   296	          <View style={styles.cardFooter}>
   297	            <View style={styles.photoBadges}>
   298	              {r.billPhoto && (
   299	                <View style={[styles.photoBadge, { backgroundColor: meta.bg }]}>
   300	                  <Feather name="image" size={11} color={meta.color} />
   301	                  <Text style={[styles.photoBadgeText, { color: meta.color }]}>Bill</Text>
   302	                </View>
   303	              )}
   304	              {r.warrantyPhoto && (
   305	                <View style={[styles.photoBadge, { backgroundColor: "#EDE9FE" }]}>
   306	                  <Feather name="shield" size={11} color="#7C3AED" />
   307	                  <Text style={[styles.photoBadgeText, { color: "#7C3AED" }]}>Warranty</Text>
   308	                </View>
   309	              )}
   310	              {!r.billPhoto && !r.warrantyPhoto && <View style={{ flex: 1 }} />}
   311	            </View>
   312	            <View style={styles.cardActions}>
   313	              <TouchableOpacity
   314	                onPress={() => confirmDelete(`/cars/${carId}/maintenance/${r.id}`, "maintenance")}
   315	                style={[styles.actionBtn, { backgroundColor: meta.bg, borderColor: meta.color }]}
   316	              >
   317	                <Feather name="trash-2" size={14} color={meta.color} />
   318	              </TouchableOpacity>
   319	            </View>
   320	          </View>
   321	        </Pressable>
   322	      );
   323	    }
   324	
   325	    if (ev.kind === "inspection") {
   326	      const r = ev.data;
   327	      return (
   328	        <Pressable
   329	          key={key}
   330	          onPress={() => { Haptics.selectionAsync(); setDetailEvent(ev); }}
   331	          style={({ pressed }) => [styles.card, { backgroundColor: C.card, borderColor: meta.color, opacity: pressed ? 0.9 : 1 }]}
   332	        >
   333	          <View style={styles.cardHeader}>
   334	            <View style={[styles.badge, { backgroundColor: meta.bg }]}>
   335	              <Feather name={meta.icon} size={11} color={meta.color} />
   336	              <Text style={[styles.badgeText, { color: meta.color }]}>{meta.label}</Text>
   337	            </View>
   338	            <Text style={[styles.dateText, { color: C.textTertiary }]}>{formatDate(r.date)}</Text>
   339	          </View>
   340	          {r.results && (
   341	            <Text style={[styles.cardTitle, { color: C.text }]} numberOfLines={3}>{r.results}</Text>
   342	          )}
   343	          {(r.place || r.cost != null) && (
   344	            <View style={styles.cardMeta}>
   345	              {r.place && (
   346	                <>
   347	                  <Feather name="map-pin" size={12} color={C.textTertiary} />
   348	                  <Text style={[styles.metaText, { color: C.textSecondary }]}>{r.place}</Text>
   349	                </>
   350	              )}
   351	              {r.cost != null && (
   352	                <>
   353	                  {r.place && <View style={styles.metaDot} />}
   354	                  <Text style={[styles.metaText, { color: C.textSecondary }]}>${r.cost.toFixed(2)}</Text>
   355	                </>
   356	              )}
   357	            </View>
   358	          )}
   359	          {r.nextInspectionDate && (
   360	            <View style={styles.cardMeta}>
   361	              <Feather name="calendar" size={12} color={C.textTertiary} />
   362	              <Text style={[styles.metaText, { color: C.textSecondary }]}>Next: {formatDate(r.nextInspectionDate)}</Text>
   363	            </View>
   364	          )}
   365	          <View style={styles.cardFooter}>
   366	            <View style={{ flex: 1 }} />
   367	            <TouchableOpacity
   368	              onPress={() => confirmDelete(`/cars/${carId}/inspections/${r.id}`, "inspections")}
   369	              style={[styles.actionBtn, { backgroundColor: meta.bg, borderColor: meta.color }]}
   370	            >
   371	              <Feather name="trash-2" size={14} color={meta.color} />
   372	            </TouchableOpacity>
   373	          </View>
   374	        </Pressable>
   375	      );
   376	    }
   377	
   378	    return null;
   379	  };
   380	
   381	  const car = carQuery.data;
   382	
   383	  if (carQuery.isLoading) {
   384	    return (
   385	      <View style={[styles.centered, { backgroundColor: C.background }]}>
   386	        <ActivityIndicator size="large" color={C.tint} />
   387	      </View>
   388	    );
   389	  }
   390	
   391	  if (!car) {
   392	    return (
   393	      <View style={[styles.centered, { backgroundColor: C.background }]}>
   394	        <Text style={{ color: C.text }}>Car not found</Text>
   395	      </View>
   396	    );
   397	  }
   398	
   399	  return (
   400	    <SwipeBackView style={{ backgroundColor: C.background }}>
   401	      <View style={[styles.header, { paddingTop: topPad + 12, borderBottomColor: C.border, backgroundColor: C.card }]}>
   402	        <View style={styles.headerRow}>
   403	          <Pressable onPress={() => router.back()} style={({ pressed }) => [styles.backBtn, { opacity: pressed ? 0.6 : 1 }]} hitSlop={8}>
   404	            <Feather name="arrow-left" size={22} color={C.text} />
   405	          </Pressable>

   410	              <Image source={{ uri: firstPhoto }} style={styles.iconBox} />

   412	              <BrandLogo
   413	                make={car.make}
   414	                size={48}
   415	                style={styles.iconBox}
   416	                fallbackBg="#DBEAFE"
   417	                fallbackIconColor="#1A56DB"

   419	            );

   433	              style={[styles.headerBtn, { backgroundColor: C.backgroundTertiary }]}

   440	              style={[styles.headerBtn, { backgroundColor: C.infoLight }]}

   442	              <Feather name="file-text" size={16} color={C.info} />
   443	            </Pressable>
   444	          </View>

   447
